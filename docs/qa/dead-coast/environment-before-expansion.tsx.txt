function placement(lon: number, lat: number, elevation?: number) {
  const frame = frameAt(lon, lat);
  return new THREE.Matrix4().compose(elevation === undefined ? frame.position : pointAt(lon, lat, elevation), frame.quaternion, new THREE.Vector3(1, 1, 1));
}

function makeTerrain() {
  // Six subdivided cube faces avoid a longitude pinch at either pole.
  const geometry = new THREE.BoxGeometry(2, 2, 2, 80, 80, 80);
  const positions = geometry.getAttribute('position');
  const colors = new Float32Array(positions.count * 3);
  const up = new THREE.Vector3();
  const color = new THREE.Color();
  for (let i = 0; i < positions.count; i++) {
    up.fromBufferAttribute(positions, i).normalize();
    const elevation = terrainHeightAt(up);
    const lat = Math.asin(up.y), lon = Math.atan2(up.x, up.z);
    const noise = Math.sin(lon * 31 + lat * 13) * Math.cos(lat * 39 - lon * 17);
    if (elevation < SEA_LEVEL + 0.08) color.set(C.wetSand);
    else if (lat < -0.3 && elevation < 0.65) color.set(C.sand);
    else if (Math.abs(lat) < 0.32) color.set(C.concrete);
    else color.set(elevation > 3.5 ? C.rock : C.grass);
    color.multiplyScalar(0.985 + noise * 0.015);
    colors.set([color.r, color.g, color.b], i * 3);
    up.multiplyScalar(RADIUS + elevation);
    positions.setXYZ(i, up.x, up.y, up.z);
  }
  geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));
  geometry.computeVertexNormals();
  geometry.computeBoundingSphere();
  return geometry;
}

function ribbon(lonStart: number, lonEnd: number, latStart: number, latEnd: number, lift = 0.026, segments = 480, fixedElevation?: number) {
  const positions: number[] = [], indices: number[] = [];
  for (let i = 0; i <= segments; i++) {
    const lon = THREE.MathUtils.lerp(lonStart, lonEnd, i / segments);
    for (const lat of [latStart, latEnd]) {
      const p = pointAt(lon, lat, (fixedElevation ?? heightAt(directionAt(lon, lat))) + lift);
      positions.push(p.x, p.y, p.z);
    }
    if (i < segments) { const a = i * 2; indices.push(a, a + 2, a + 1, a + 2, a + 3, a + 1); }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  g.setIndex(indices); g.computeVertexNormals();
  return g;
}

function makeStreets() {
  const batch = new SceneryBatch(), identity = new THREE.Matrix4();
  const add = (g: THREE.BufferGeometry, color: string) => { batch.add(g, identity, color); g.dispose(); };
  const band = (a: number, b: number, color: string, h: number) => add(ribbon(-Math.PI, Math.PI, a, b, 0.018, 640, h), color);
  band(-0.064, 0.064, '#697b7a', 0.25);
  band(-0.116, -0.064, C.plaster, 0.45); band(0.064, 0.116, C.plaster, 0.45);
  band(-0.069, -0.064, C.bone, 0.456); band(0.064, 0.069, C.bone, 0.456);
  band(-0.059, -0.057, C.bone, 0.253); band(0.057, 0.059, C.bone, 0.253);
  for (let i = 0; i < 232; i++) {
    const lon = i / 232 * Math.PI * 2 - Math.PI;
    for (const side of [-1, 1]) {
      batch.shape(box, placement(lon, side * 0.09), [0, 0.023, 0], [0.022, 0.008, 1.77], '#a5aa9a');
      batch.shape(box, placement(lon, side * 0.064, 0.45), [0, -0.1, 0], [1.01, 0.2, 0.065], C.bone);
    }
    if (i % 5 === 0) batch.shape(box, placement(lon, 0), [0, 0.029, 0], [0.78, 0.012, 0.095], C.bone);
  }
  for (const cross of [-0.45, 0.12, 1.28, 2.8, -2.1]) for (let i = 0; i < 7; i++) {
    batch.shape(box, placement(cross + (i - 3) * 0.016, 0), [0, 0.033, 0], [0.32, 0.012, 3.72], C.bone);
  }
  // Paths fit through the actual gaps between buildings, descending to coves.
  for (const lon of [-0.6, 0.30, 0.69, 1.88, -1.89]) {
    for (let i = 0; i < 24; i++) {
      const lat = -0.117 - i * 0.012;
      const bend = Math.sin(i / 24 * Math.PI) * 0.024;
      if (terrainHeightAt(directionAt(lon + bend, lat)) <= SEA_LEVEL + 0.06) break;
      add(ribbon(lon + bend - 0.033, lon + bend + 0.033, lat - 0.012, lat, 0.025, 8), i < 10 ? '#b2b3a2' : '#c4bd9b');
    }
  }
  // Exact raised treads match the controller's staircase height function.
  for (let i = 0; i < STAIRS.count; i++) {
    const lat = STAIRS.startLat + i * STAIRS.treadAngle;
    const h = STAIRS.baseHeight + (i + 1) * STAIRS.rise;
    const half = STAIRS.width / (2 * RADIUS);
    add(ribbon(STAIRS.lon - half, STAIRS.lon + half, lat, lat + STAIRS.treadAngle, 0.008, 12, h), C.plaster);
    add(ribbon(STAIRS.lon - half, STAIRS.lon + half, lat, lat + 0.0018, 0.014, 12, h), C.bone);
    const m = placement(STAIRS.lon, lat + STAIRS.treadAngle / 2, h);
    batch.shape(box, m, [0, -STAIRS.rise / 2, 0], [STAIRS.width, STAIRS.rise, STAIRS.treadAngle * RADIUS], C.plaster);
    for (const side of [-1, 1]) {
      const edge = placement(STAIRS.lon + side * (half + 0.003), lat, h);
      batch.shape(box, edge, [0, -0.15, 0], [0.18, 0.32, STAIRS.treadAngle * RADIUS + 0.1], C.bone);
      if (i % 3 === 0) batch.shape(box, edge, [0, 0.47, 0], [0.065, 0.94, 0.065], C.graphite);
    }
  }
  for (const side of [-1, 1]) for (let i = 0; i < STAIRS.count - 1; i++) {
    const lon = STAIRS.lon + side * (STAIRS.width / (2 * RADIUS) + 0.003);
    const lat = STAIRS.startLat + i * STAIRS.treadAngle;
    const start = pointAt(lon, lat, STAIRS.baseHeight + (i + 1) * STAIRS.rise + 0.94);
    const end = pointAt(lon, lat + STAIRS.treadAngle, STAIRS.baseHeight + (i + 2) * STAIRS.rise + 0.94);
    const delta = end.clone().sub(start);
    const matrix = new THREE.Matrix4().compose(start.clone().lerp(end, 0.5), new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0, 1, 0), delta.clone().normalize()), new THREE.Vector3(0.038, delta.length(), 0.038));
    batch.add(cylinder, matrix, C.graphite);
  }
  return batch.finish();
}

function makeCoast() {
  const positions: number[] = [], indices: number[] = [], colors: number[] = [];
  const color = new THREE.Color(), segments = 480;
  for (let i = 0; i <= segments; i++) {
    const lon = i / segments * Math.PI * 2 - Math.PI;
    let shoreLat = -0.2;
    while (shoreLat > -1.52 && terrainHeightAt(directionAt(lon, shoreLat)) > SEA_LEVEL) shoreLat -= 0.005;
    for (let j = 0; j < 4; j++) {
      const p = pointAt(lon, shoreLat - j * 0.014, SEA_LEVEL + 0.014 + j * 0.001);
      positions.push(p.x, p.y, p.z);
      color.set(['#d0d9bd', '#a4c1b7', '#83aaa7', '#67949a'][j]); colors.push(color.r, color.g, color.b);
    }
    if (i < segments) for (let j = 0; j < 3; j++) { const a = i * 4 + j; indices.push(a, a + 1, a + 4, a + 1, a + 5, a + 4); }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  g.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
  g.setIndex(indices); g.computeVertexNormals(); return g;
}

  sign(beach, 'COASTAL FREQUENCY', 'A SIGNAL FROM SOMEWHERE ELSE', [-0.32, 1.36, 0.04], 1.42, 0.52, Math.PI);
  b(beach, [-0.32, 0.93, 0.105], [0.065, 0.86, 0.065], C.rust);
  const lookout = placement(2.7, 0.83);
  for (const x of [-1.5, 1.5]) for (const z of [-1.5, 1.5]) b(lookout, [x, 1.47, z], [0.15, 2.94, 0.15], C.rust);
  b(lookout, [0, 3.02, 0], [3.55, 0.17, 3.55], C.graphite);
  for (const x of [-1.5, 1.5]) b(lookout, [x, 0.94, 0], [0.1, 0.1, 3.04], C.bone);
  b(lookout, [0, 0.94, -1.5], [3.04, 0.1, 0.1], C.bone);
  sign(lookout, 'KEEP WANDERING', 'THERE IS ALWAYS ANOTHER SIDE', [0, 2.48, 1.585], 2.67, 0.61, 0, true);
  const wires = new THREE.BufferGeometry(); wires.setAttribute('position', new THREE.Float32BufferAttribute(wirePoints, 3));
  return { architecture: architecture.finish(), greenery: greenery.finish(), stones: stones.finish(), illuminated: illuminated.finish(), wires, signs };
}

function Sign({ title, subtitle, matrix, width, height, dark = false }: SignData) {
  const texture = useMemo(() => {
    const canvas = document.createElement('canvas'); canvas.width = 1024; canvas.height = Math.max(128, Math.round(1024 * height / width));
    const ctx = canvas.getContext('2d')!;
    ctx.fillStyle = dark ? C.graphite : C.bone; ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = dark ? C.bone : C.graphite; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.font = `900 ${canvas.height * (subtitle ? 0.5 : 0.66)}px Arial, sans-serif`;
    ctx.fillText(title, 512, canvas.height * (subtitle ? 0.4 : 0.52), 950);
    if (subtitle) { ctx.font = `600 ${canvas.height * 0.12}px Arial, sans-serif`; ctx.fillText(subtitle, 512, canvas.height * 0.83, 900); }
    const result = new THREE.CanvasTexture(canvas); result.colorSpace = THREE.SRGBColorSpace; result.anisotropy = 4; return result;
  }, [dark, height, subtitle, title, width]);
  useEffect(() => () => texture.dispose(), [texture]);
  return <mesh matrix={matrix} matrixAutoUpdate={false}><planeGeometry args={[width, height]} /><meshStandardMaterial map={texture} roughness={1} /></mesh>;
}

const waterVertex = `
  varying vec3 vWorld;
  varying vec3 vNormal;
  void main() {
    vec4 world = modelMatrix * vec4(position, 1.0);
    vWorld = world.xyz; vNormal = normalize(mat3(modelMatrix) * normal);
    gl_Position = projectionMatrix * viewMatrix * world;
  }
`;
const waterFragment = `
  uniform float time;
  varying vec3 vWorld;
  varying vec3 vNormal;
  void main() {
    vec3 n = normalize(vNormal);
    float ripple = sin(vWorld.x * 1.8 + vWorld.z * 0.55 + time * 0.28) * sin(vWorld.y * 1.6 - vWorld.z * 0.9 - time * 0.17);
    float fine = sin(vWorld.x * 5.7 + vWorld.y * 3.8 + time * 0.36) * sin(vWorld.z * 4.4 - vWorld.y * 2.1 - time * 0.22);
    float facing = max(dot(n, normalize(cameraPosition - vWorld)), 0.0);
    float light = 0.78 + 0.22 * max(dot(n, normalize(vec3(0.6, 0.8, 1.0))), 0.0);
    vec3 water = mix(vec3(0.055, 0.145, 0.18), vec3(0.20, 0.37, 0.39), pow(1.0 - facing, 2.0));
    water += ripple * 0.005; water += smoothstep(0.88, 0.98, fine) * 0.035;
    gl_FragColor = vec4(water * light, 1.0);
    #include <tonemapping_fragment>
    #include <colorspace_fragment>
  }
`;
function Ocean() {
  const material = useRef<THREE.ShaderMaterial>(null), uniforms = useMemo(() => ({ time: { value: 0 } }), []);
  useFrame((_, dt) => { if (material.current) material.current.uniforms.time.value += Math.min(dt, 0.05); });
  return <mesh><sphereGeometry args={[RADIUS + SEA_LEVEL, 128, 80]} /><shaderMaterial ref={material} uniforms={uniforms} vertexShader={waterVertex} fragmentShader={waterFragment} /></mesh>;
}

/** A closed planet: every district and coastline shares the walking surface. */
export default function Environment() {
  const world = useMemo(() => ({ terrain: makeTerrain(), streets: makeStreets(), coast: makeCoast(), ...buildScenery() }), []);
  useEffect(() => () => { for (const value of Object.values(world)) if (value instanceof THREE.BufferGeometry) value.dispose(); }, [world]);
  return (
    <group>
      <mesh geometry={world.terrain} receiveShadow><meshStandardMaterial vertexColors roughness={1} /></mesh>
      <mesh geometry={world.streets} receiveShadow><meshStandardMaterial vertexColors roughness={1} side={THREE.DoubleSide} /></mesh>
      <Ocean />
      <mesh geometry={world.coast}><meshStandardMaterial vertexColors roughness={0.94} side={THREE.DoubleSide} /></mesh>
      <mesh geometry={world.architecture} castShadow receiveShadow userData={{ cameraObstacle: true }}><meshStandardMaterial vertexColors roughness={0.95} /></mesh>
      <mesh geometry={world.greenery} castShadow receiveShadow><meshStandardMaterial vertexColors roughness={1} flatShading /></mesh>
      <mesh geometry={world.stones} castShadow receiveShadow><meshStandardMaterial vertexColors roughness={1} flatShading /></mesh>
      <mesh geometry={world.illuminated}><meshStandardMaterial vertexColors roughness={0.78} emissive='#998668' emissiveIntensity={0.12} /></mesh>
      <lineSegments geometry={world.wires}><lineBasicMaterial color='#545e55' /></lineSegments>
      {world.signs.map((s, i) => <Sign key={i} {...s} />)}
    </group>
  );
}
