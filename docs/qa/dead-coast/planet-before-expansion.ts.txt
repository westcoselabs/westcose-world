import { Matrix4, Quaternion, Vector3 } from 'three';

/** Layout v2: a fixed planet, radial up/gravity, metres. No planar wrapping. */
export const RADIUS = 36;
export const SEA_LEVEL = -.8;
export const PLANET_VERSION = 2;
export const ROAD_HALF_LAT = .064;
export const SIDEWALK_OUTER_LAT = .116;
export const SPAWN_COORDS = { lon: -.35, lat: 0 };
export const clamp = (v: number, low: number, high: number) => Math.max(low, Math.min(high, v));
export function smoothstep(low: number, high: number, value: number) {
  const t = clamp((value - low) / (high - low), 0, 1);
  return t * t * (3 - 2 * t);
}
export const angleDifference = (a: number, b: number) => Math.atan2(Math.sin(a - b), Math.cos(a - b));
export function directionAt(lon: number, lat: number): Vector3 {
  return new Vector3(Math.sin(lon) * Math.cos(lat), Math.sin(lat), Math.cos(lon) * Math.cos(lat));
}
export function coordinatesAt(d: { x: number; y: number; z: number }) {
  const length = Math.hypot(d.x, d.y, d.z) || 1;
  return { lon: Math.atan2(d.x, d.z), lat: Math.asin(clamp(d.y / length, -1, 1)) };
}
export const STAIRS = { lon: .27, startLat: .116, endLat: .326, width: 2.6, count: 12, rise: .22, treadAngle: .0175, baseHeight: .25 };
/** Continuous substrate; raised pavements/stairs are separate scene surfaces. */
export function terrainHeightAt(d: { x: number; y: number; z: number }): number {
  const { lon, lat } = coordinatesAt(d);
  const coast = -.48 + .1 * Math.sin(lon * 3) + .07 * Math.cos(lon * 5);
  const land = smoothstep(coast - .14, coast + .19, lat);
  // Longitude becomes undefined at a pole; fade longitude-dependent relief
  // into a single cap height so approaching from any bearing is continuous.
  const polarBlend = 1 - smoothstep(1.1, 1.48, Math.abs(lat));
  const hills = smoothstep(.3, .78, lat) * (2.25 + polarBlend * (-.95 + 1.9 * Math.pow(Math.sin(lon * 2 + lat * 3), 2)));
  const ridge = 2.2 * smoothstep(.85, 1.25, lat);
  const rolling = .48 * Math.pow(Math.sin(lon * 3) * Math.sin(lat * 5), 2) * smoothstep(.26, .48, Math.abs(lat)) * polarBlend;
  let h = -2.8 + land * (3.05 + hills + ridge + rolling);
  const terrace = (1 - smoothstep(.11, .19, Math.abs(angleDifference(lon, STAIRS.lon)))) * smoothstep(.29, .326, lat) * (1 - smoothstep(.5, .6, lat));
  if (terrace > 0) h = Math.max(h, .25 + terrace * STAIRS.count * STAIRS.rise);
  return h;
}
export function heightAt(d: { x: number; y: number; z: number }): number {
  const { lon, lat } = coordinatesAt(d);
  if (Math.abs(lat) <= ROAD_HALF_LAT) return .25;
  if (Math.abs(lat) <= SIDEWALK_OUTER_LAT) return .45;
  if (lat > STAIRS.startLat && lat < STAIRS.endLat && Math.abs(angleDifference(lon, STAIRS.lon)) < STAIRS.width / (2 * RADIUS)) {
    return STAIRS.baseHeight + Math.min(STAIRS.count, Math.floor((lat - STAIRS.startLat) / STAIRS.treadAngle) + 1) * STAIRS.rise;
  }
  return terrainHeightAt(d);
}
export const waterAt = (d: { x: number; y: number; z: number }) => terrainHeightAt(d) < SEA_LEVEL;
export function pointAt(lon: number, lat: number, elevation?: number): Vector3 {
  const up = directionAt(lon, lat);
  return up.multiplyScalar(RADIUS + (elevation ?? Math.max(SEA_LEVEL, heightAt(up))));
}
export function frameAt(lon: number, lat: number) {
  const up = directionAt(lon, lat);
  const east = new Vector3(Math.cos(lon), 0, -Math.sin(lon));
  const north = new Vector3(-Math.sin(lat) * Math.sin(lon), Math.cos(lat), -Math.sin(lat) * Math.cos(lon));
  const quaternion = new Quaternion().setFromRotationMatrix(new Matrix4().makeBasis(east, up, north.clone().negate()));
  return { position: pointAt(lon, lat), quaternion, up, east, north };
}

export function areaAt(d: { x: number; y: number; z: number }): string {
  const {lon,lat}=coordinatesAt(d);
  if(waterAt(d))return 'Open Water';
  if(lat<-.29)return 'The Coast';
  if(lat>.62)return 'The High Ground';
  if(lat>.29)return 'Overlook Steps';
  if(Math.abs(lon)<.8)return 'Studio Lane';
  if(lon>.8&&lon<2)return 'Maker’s Row';
  if(Math.abs(lon)>=2)return 'The Far Side';
  return 'West Passage';
}
